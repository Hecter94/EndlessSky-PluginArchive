    invisible
    non-blocking
    transition
    repeat
    to offer
        has "main plot plus: enabled main plot soft block"
        not "main plot completed"
        not "event: main plot plus: postverta reveal patch - hidden"
    to accept
        has "event: remnant: postverta reveal"
    destination Ruin
    on accept
        event "main plot plus: postverta reveal patch - hidden"

    non-blocking
