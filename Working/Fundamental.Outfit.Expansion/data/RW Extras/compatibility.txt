mission "FOE: RW Combined Human Missile Turrets"
	landing
	invisible
	non-blocking
	repeat
	to offer
		has "installed plugin: Combined Human Missile Turrets"
	on offer
		event "foe: rw combined human missile turrets installed"
		fail

event "foe: rw combined human missile turrets installed"
	outfitter "Delta V Advanced"
		remove "Heavy Rocket Turret"
		remove "Meteor Missile Turret"
	outfitter "Lovelace Advanced"
		remove "Sidewinder Missile VLS"
	outfitter "Syndicate Advanced"
		remove "Meteor Missile Turret"
